**Architecture:** Add `media` table with status-aware CHECK constraints and a `partial UNIQUE(babyId, contentHash) WHERE status='ready'`. Add `media` FK on the existing `entry_media.mediaId` (column shipped in P2a, FK deferred). All media routes share three primitives: (a) the `withAuthorizedResource` template extended for `(action × media-status × baby-status)` gates per §6.3.1, (b) a service module `lib/media/*` that owns the filesystem and is the only place that touches `data/media/`, (c) `loadAndAssertTarget` extended with a `'media'` case that already JOINs `babies` to enforce parent-chain visibility. Upload is one big request handler that streams multipart, hashes in flight, sniffs by magic-bytes, and only enters the dedupe branch after the server has the bytes. Reconcile runs once at boot and on a 24h interval inside `instrumentation.node.ts`.

**Tech Stack:** `sharp` (image derive + photo metadata), `file-type` (magic-byte MIME sniff), `ffmpeg-static` + `ffprobe-static` + `fluent-ffmpeg` (video probe + poster frame), `formidable` (streaming multipart — Next.js App Router `req.formData()` buffers the whole body and is unusable for 500MB videos), `crypto.createHash('sha256')` streaming, Drizzle migrations for the new table and FK.

```ts
// lib/media/paths.ts
import { join, resolve, sep } from 'node:path';

export type Variant = 'original' | 'large' | 'thumb' | 'poster';

export function stagingDir(dataDir: string, uploadId: string): string {
  if (uploadId.includes('/') || uploadId.includes('\\') || uploadId.includes('..')) {
    throw new Error('invalid_upload_id');
  }
  return join(dataDir, 'media', '_staging', uploadId);
}

export function finalRelativePath(babyId: string, tsMs: number, mediaId: string): string {
  const d = new Date(tsMs);
  const year = String(d.getUTCFullYear());
  const month = String(d.getUTCMonth() + 1).padStart(2, '0');
  return `media/${babyId}/${year}/${month}/${mediaId}`;
}

export function finalDirFromRelative(dataDir: string, relativePath: string): string {
  const abs = resolve(dataDir, relativePath);
  const rootAbs = resolve(dataDir, 'media');
  if (!abs.startsWith(rootAbs + sep) && abs !== rootAbs) {
    throw new Error('path_escape');
  }
  return abs;
}

export function variantFilename(variant: Variant, originalExt: string): string {
  return variant === 'original' ? `original.${originalExt}` : `${variant}.webp`;
}

export function resolveVariantPath(
  dataDir: string,
  relativePath: string,
  variant: Variant,
  originalExt: string
): string {
  return join(finalDirFromRelative(dataDir, relativePath), variantFilename(variant, originalExt));
}
```

```ts
// lib/media/hash.ts
import { createHash, type Hash } from 'node:crypto';
import { Readable, Writable } from 'node:stream';

export interface HashResult {
  hash: string;
  bytes: number;
}

export class MediaTooLargeError extends Error {
  code = 'too_large' as const;
}

export async function pipeThroughHasher(
  src: Readable,
  sink: Writable,
  opts: { maxBytes: number }
): Promise<HashResult> {
  const hasher: Hash = createHash('sha256');
  let bytes = 0;

  return await new Promise<HashResult>((resolve, reject) => {
    const onErr = (e: Error) => {
      src.destroy();
      sink.destroy();
      reject(e);
    };
    src.on('error', onErr);
    sink.on('error', onErr);
    sink.on('finish', () => resolve({ hash: hasher.digest('hex'), bytes }));
    src.on('data', (chunk: Buffer) => {
      bytes += chunk.length;
      if (bytes > opts.maxBytes) return onErr(new MediaTooLargeError('too_large'));
      hasher.update(chunk);
      if (!sink.write(chunk)) src.pause();
    });
    sink.on('drain', () => src.resume());
    src.on('end', () => sink.end());
  });
}
```

```ts
// lib/media/sniff.ts
import { fileTypeFromFile } from 'file-type';
import sharp from 'sharp';
import ffprobePath from 'ffprobe-static';
import ffmpeg from 'fluent-ffmpeg';

ffmpeg.setFfprobePath(ffprobePath.path);

export class MediaUnsupportedError extends Error {
  code = 'unsupported_media' as const;
}

const PHOTO_MIMES = new Set([
  'image/jpeg',
  'image/png',
  'image/webp',
  'image/heic',
  'image/heif'
]);
const VIDEO_MIMES = new Set(['video/mp4', 'video/quicktime']);

const EXT_BY_MIME: Record<string, string> = {
  'image/jpeg': 'jpg',
  'image/png': 'png',
  'image/webp': 'webp',
  'image/heic': 'heic',
  'image/heif': 'heif',
  'video/mp4': 'mp4',
  'video/quicktime': 'mov'
};

export interface SniffResult {
  type: 'photo' | 'video';
  mimeType: string;
  ext: string;
  width: number;
  height: number;
  durationSec: number | null;
  takenAtMs: number | null;
}

export async function sniffAndValidate(absPath: string): Promise<SniffResult> {
  const ft = await fileTypeFromFile(absPath);
  if (!ft) throw new MediaUnsupportedError('no_magic_bytes');
  const mime = ft.mime;

  if (PHOTO_MIMES.has(mime)) {
    const meta = await sharp(absPath, { failOn: 'error' }).metadata().catch(() => null);
    if (!meta || !meta.width || !meta.height) throw new MediaUnsupportedError('photo_metadata_failed');
    const exifMs = readExifTakenAtMs(meta);
    return {
      type: 'photo',
      mimeType: mime,
      ext: EXT_BY_MIME[mime],
      width: meta.width,
      height: meta.height,
      durationSec: null,
      takenAtMs: exifMs
    };
  }
  if (VIDEO_MIMES.has(mime)) {
    const probe = await new Promise<{ w: number; h: number; d: number }>((resolve, reject) => {
      ffmpeg.ffprobe(absPath, (err, data) => {
        if (err) return reject(err);
        const v = data.streams?.find((s) => s.codec_type === 'video');
        if (!v?.width || !v?.height) return reject(new MediaUnsupportedError('video_no_video_stream'));
        const d = Number(data.format?.duration ?? 0);
        resolve({ w: v.width, h: v.height, d });
      });
    }).catch((e) => {
      if (e instanceof MediaUnsupportedError) throw e;
      throw new MediaUnsupportedError('video_probe_failed');
    });
    return {
      type: 'video',
      mimeType: mime,
      ext: EXT_BY_MIME[mime],
      width: probe.w,
      height: probe.h,
      durationSec: Math.max(1, Math.round(probe.d)),
      takenAtMs: null
    };
  }
  throw new MediaUnsupportedError(`mime_not_allowed:${mime}`);
}

function readExifTakenAtMs(meta: sharp.Metadata): number | null {
  const exif = meta.exif;
  if (!exif) return null;
  // EXIF DateTimeOriginal is in the binary blob — sharp does not parse it.
  // Lightweight regex scan: "YYYY:MM:DD HH:MM:SS" in the binary buffer.
  const s = exif.toString('latin1');
  const m = s.match(/(\d{4}):(\d{2}):(\d{2}) (\d{2}):(\d{2}):(\d{2})/);
  if (!m) return null;
  const [, y, mo, d, h, mi, se] = m;
  const t = Date.UTC(+y, +mo - 1, +d, +h, +mi, +se);
  return Number.isFinite(t) ? t : null;
}
```

```ts
export async function deriveVideoPoster(
  originalAbs: string,
  outDir: string
): Promise<string> {

  // 1. Extract frame at t=0 into a raw .jpg in the same dir, then re-encode to webp via sharp.
  const tmp = join(outDir, 'poster.jpg');
  await new Promise<void>((res, rej) => {

      .on('error', rej)
      .run();
```

```ts
// lib/media/storage.ts
import { mkdir, rename, rm } from 'node:fs/promises';
import { join } from 'node:path';
import {
  stagingDir,
  finalRelativePath,
  finalDirFromRelative,
  variantFilename
} from './paths';

export interface StagingHandle {
  stagingDir: string;
  rawPath: string;
}

export async function prepareStaging(
  dataDir: string,
  uploadId: string
): Promise<StagingHandle> {
  const dir = stagingDir(dataDir, uploadId);
  await mkdir(dir, { recursive: true });
  return { stagingDir: dir, rawPath: join(dir, 'raw.bin') };
}

export interface CommitOpts {
  mediaId: string;
  babyId: string;
  createdAtMs: number;
  ext: string;
  kind: 'photo' | 'video';
}

export async function commitStaging(dataDir: string, opts: CommitOpts): Promise<string> {
  const rel = finalRelativePath(opts.babyId, opts.createdAtMs, opts.mediaId);
  const finalDir = finalDirFromRelative(dataDir, rel);
  await mkdir(finalDir, { recursive: true });

  const stage = stagingDir(dataDir, opts.mediaId);
  const moves: Array<[string, string]> = [
    [join(stage, variantFilename('original', opts.ext)), join(finalDir, variantFilename('original', opts.ext))]
  ];
  if (opts.kind === 'photo') {
    moves.push([join(stage, 'large.webp'), join(finalDir, 'large.webp')]);
    moves.push([join(stage, 'thumb.webp'), join(finalDir, 'thumb.webp')]);
  } else {
    moves.push([join(stage, 'poster.webp'), join(finalDir, 'poster.webp')]);
  }
  for (const [from, to] of moves) {
    await rename(from, to);
  }
  await rm(stage, { recursive: true, force: true });
  return rel;
}

export async function purgeStagingDir(dir: string): Promise<void> {
  await rm(dir, { recursive: true, force: true });
}

export async function purgeFinalDir(dataDir: string, relativePath: string): Promise<void> {
  const abs = finalDirFromRelative(dataDir, relativePath); // throws on escape
  await rm(abs, { recursive: true, force: true });
}
```

**Important:** `commitStaging` uses the **mediaId as the staging-dir name**. This is intentional: after §6.2 step 3 the server has chosen the mediaId, and rewriting the staging dir name would not be atomic across renames. The upload pipeline (Task 10) always passes the chosen `mediaId` (uuid) to `prepareStaging`.

```ts
// lib/media/output.ts
import { variantFilename, type Variant } from './paths';

export class OutputBadRequestError extends Error {
  code = 'bad_size' as const;
}

export interface MediaForOutput {
  id: string;
  type: 'photo' | 'video';
  mimeType: string;
  originalExt: string;
  filename: string;
}

export interface OutputDescriptor {
  variant: Variant;
  variantFile: string;
  contentType: string;
  contentDisposition: string;
}

const ALLOWED: ReadonlyArray<Variant> = ['original', 'large', 'thumb', 'poster'];

export function resolveOutputVariant(
  row: MediaForOutput,
  rawSize: string
): OutputDescriptor {
  if (!(ALLOWED as readonly string[]).includes(rawSize)) {
    throw new OutputBadRequestError('illegal_size');
  }
  let variant = rawSize as Variant;
  if (row.type === 'photo' && variant === 'poster') {
    throw new OutputBadRequestError('photo_has_no_poster');
  }
  if (row.type === 'video' && (variant === 'thumb' || variant === 'large')) {
    variant = 'poster';
  }

  const baseFilename = sanitizeForHeader(row.filename);
  const file = variantFilename(variant, row.originalExt);
  if (variant === 'original') {
    return {
      variant,
      variantFile: file,
      contentType: row.mimeType,
      contentDisposition: `attachment; filename*=UTF-8''${rfc5987Encode(baseFilename)}`
    };
  }
  return {
    variant,
    variantFile: file,
    contentType: 'image/webp',
    contentDisposition: `inline; filename*=UTF-8''${rfc5987Encode(baseFilename + '.webp')}`
  };
}

function sanitizeForHeader(s: string): string {
  return s.replace(/[ -"\\\r\n]/g, '_').slice(0, 200) || 'media';
}

function rfc5987Encode(s: string): string {
  return encodeURIComponent(s).replace(/['()*]/g, (c) => '%' + c.charCodeAt(0).toString(16).toUpperCase());
}
```

```ts
import { babies, entries, media } from '@/lib/db/schema';
import type { Action, PermissionResource } from './actions';
import { NotFoundError } from './errors';
import { assertPermission } from './assert';
import { UUID_RE } from './responses';

export interface LoadAndAssertOptions {
  id: string;
  table: 'babies' | 'entries' | 'media' | 'milestones' | 'users';
  allowedStatuses?: string[];
  allowedParentStatuses?: string[];
  requirePermission: { userId: string; action: Action };
  toResource?: (row: any) => PermissionResource;
  dataDir?: string;
}

export async function loadAndAssertTarget<R = unknown>(
  opts: LoadAndAssertOptions
): Promise<R> {
  if (!UUID_RE.test(opts.id)) throw new NotFoundError(opts.table);

  const dataDir = opts.dataDir
    ? resolve(opts.dataDir)
    : process.env.BABYLOOM_DATA_DIR
      ? resolve(process.env.BABYLOOM_DATA_DIR)
      : resolve(process.cwd(), 'data');

  const { db } = getDb({ dataDir });

  let row: any;
  switch (opts.table) {
    case 'babies':
      row = db.select().from(babies).where(eq(babies.id, opts.id)).get();
      break;
    case 'entries':
      row = db
        .select({
          id: entries.id,
          babyId: entries.babyId,
          authorId: entries.authorId,
          content: entries.content,
          occurredAt: entries.occurredAt,
          status: entries.status,
          createdAt: entries.createdAt,
          updatedAt: entries.updatedAt,
          deletedAt: entries.deletedAt,
          deletedBy: entries.deletedBy,
          babyStatus: babies.status
        })
        .from(entries)
        .innerJoin(babies, eq(babies.id, entries.babyId))
        .where(eq(entries.id, opts.id))
        .get();
      if (row) {
        const okParent = (opts.allowedParentStatuses ?? ['active']).includes(row.babyStatus);
        if (!okParent) row = null;
      }
      break;
    case 'media':
      row = db
        .select({
          id: media.id,
          babyId: media.babyId,
          uploadedBy: media.uploadedBy,
          status: media.status,
          type: media.type,
          mimeType: media.mimeType,
          relativePath: media.relativePath,
          originalExt: media.originalExt,
          filename: media.filename,
          sizeBytes: media.sizeBytes,
          contentHash: media.contentHash,
          width: media.width,
          height: media.height,
          durationSec: media.durationSec,
          createdAt: media.createdAt,
          updatedAt: media.updatedAt,
          deletedAt: media.deletedAt,
          deletedBy: media.deletedBy,
          clientUploadId: media.clientUploadId,
          babyStatus: babies.status
        })
        .from(media)
        .innerJoin(babies, eq(babies.id, media.babyId))
        .where(eq(media.id, opts.id))
        .get();
      if (row) {
        const okParent = (opts.allowedParentStatuses ?? ['active']).includes(row.babyStatus);
        if (!okParent) row = null;
      }
      break;
    case 'milestones': {
      const { milestones } = await import('@/lib/db/schema');
      row = db.select().from(milestones).where(eq(milestones.id, opts.id)).get();
      break;
    }
    case 'users': {
      const { users } = await import('@/lib/db/schema');
      row = db.select().from(users).where(eq(users.id, opts.id)).get();
      break;
    }
    default:
      throw new Error(`unsupported table: ${opts.table}`);
  }

  if (!row) throw new NotFoundError(opts.table);

  if (opts.allowedStatuses && !opts.allowedStatuses.includes(row.status)) {
    throw new NotFoundError(opts.table);
  }

  const defaultResource = (target: any): PermissionResource => {
    switch (opts.table) {
      case 'babies':
        return { babyId: target.id };
      case 'entries':
        return {
          babyId: target.babyId,
          entryId: target.id,
          authorId: target.authorId,
          deletedBy: target.deletedBy ?? undefined
        };
      case 'media':
        return {
          babyId: target.babyId,
          mediaId: target.id,
          uploadedBy: target.uploadedBy,
          deletedBy: target.deletedBy ?? undefined
        };
      case 'milestones':
        return {};
      case 'users':
        return { targetUserId: target.id };
    }
  };
  const resource = opts.toResource ? opts.toResource(row) : defaultResource(row);
  await assertPermission(opts.requirePermission.userId, opts.requirePermission.action, resource, {
    dataDir
  });

  return row as R;
}
```

## Task 10 — `lib/media/upload-pipeline.ts`: the orchestrator

**Files:**
- Create: `lib/media/upload-pipeline.ts`

This module is **server-only** and pure-ish: it takes `(dataDir, db, userId, parsed multipart fields, the open file path of the staged raw upload)`, performs steps 2.5 → 6 of §6.2, and returns one of `{ kind: 'created' | 'deduped' | 'retried', mediaId, status, deduplicated }`. The route handler in Task 11 owns the multipart parse and translation to HTTP responses.

- [ ] **Step 1: Implement**

```ts
// lib/media/upload-pipeline.ts
import { and, eq, ne } from 'drizzle-orm';
import { randomUUID, createHash } from 'node:crypto';
import { createReadStream, createWriteStream } from 'node:fs';
import { rename, stat, unlink } from 'node:fs/promises';
import { join } from 'node:path';
import { getDb } from '@/lib/db/client';
import { entryMedia, media } from '@/lib/db/schema';
import { pipeThroughHasher } from './hash';
import { sniffAndValidate, MediaUnsupportedError } from './sniff';
import { prepareStaging, commitStaging, purgeStagingDir } from './storage';
import { derivePhotoVariants, deriveVideoPoster } from './derive';
import { variantFilename } from './paths';
import { getMediaLimits } from './config';

export type UploadOutcome =
  | { kind: 'retry'; mediaId: string; status: 'pending' | 'processing'; pollUrl: string }
  | { kind: 'ready_idempotent'; mediaId: string; deduplicated: false }
  | { kind: 'deduped'; mediaId: string }
  | { kind: 'created'; mediaId: string };

export class UploadValidationError extends Error {
  constructor(public httpStatus: number, public errCode: string) {
    super(errCode);
  }
}

export interface UploadInput {
  dataDir: string;
  userId: string;
  babyId: string;           // already loaded + authorized by route
  entryId: string | null;   // already loaded + authorized by route (if provided)
  clientUploadId: string;
  filenameRaw: string;
  uploadedFilePath: string; // path where the multipart parser wrote the user's bytes
  uploadedSizeBytes: number;
}

export async function runUploadPipeline(input: UploadInput): Promise<UploadOutcome> {
  const { db } = getDb({ dataDir: input.dataDir });
  const limits = getMediaLimits(input.dataDir);

  // §6.2 step 2.5 — match prior clientUploadId for THIS user.
  const prior = db
    .select({ id: media.id, status: media.status, uploadedBy: media.uploadedBy })
    .from(media)
    .where(and(eq(media.clientUploadId, input.clientUploadId), eq(media.uploadedBy, input.userId)))
    .get();

  if (prior) {
    if (prior.status === 'pending' || prior.status === 'processing') {
      return { kind: 'retry', mediaId: prior.id, status: prior.status, pollUrl: `/api/media/${prior.id}/status` };
    }
    if (prior.status === 'ready') {
      return { kind: 'ready_idempotent', mediaId: prior.id, deduplicated: false };
    }
    // failed → fall through and start a new row
  }

  // Step 3 — new row in pending, then stream-hash-sniff.
  const mediaId = randomUUID();
  const nowMs = Date.now();
  const sanitizedFilename = sanitizeFilenameDisplay(input.filenameRaw);
  db.insert(media).values({
    id: mediaId,
    babyId: input.babyId,
    uploadedBy: input.userId,
    clientUploadId: input.clientUploadId,
    status: 'pending',
    filename: sanitizedFilename,
    createdAt: nowMs,
    updatedAt: nowMs
  }).run();

  const staging = await prepareStaging(input.dataDir, mediaId);
  // Move the parser's temp file into our staging dir as raw.bin.
  await rename(input.uploadedFilePath, staging.rawPath);

  // Server-authoritative size check (parser may have over-collected).
  const sz = await stat(staging.rawPath);
  // We don't yet know photo vs video — use the higher cap; the type-specific cap is rechecked after sniff.
  if (sz.size > Math.max(limits.maxPhotoBytes, limits.maxVideoBytes)) {
    await purgeStagingDir(staging.stagingDir);
    await markFailed(db, mediaId, 'too_large');
    throw new UploadValidationError(413, 'too_large');
  }

  // Server-authoritative streaming hash. The parser already wrote to disk, so we re-hash
  // the staged file in one pass (the spec's "stream-while-receive" optimisation is moot
  // here because formidable's temp file is already on the same disk and only one fs call away).
  const hash = await streamHashFile(staging.rawPath);

  // MIME sniff + dimensions/duration.
  let sniffed;
  try {
    sniffed = await sniffAndValidate(staging.rawPath);
  } catch (e) {
    await purgeStagingDir(staging.stagingDir);
    await markFailed(db, mediaId, 'unsupported_media');
    if (e instanceof MediaUnsupportedError) throw new UploadValidationError(422, e.message);
    throw e;
  }

  // Type-specific size enforcement.
  const cap = sniffed.type === 'photo' ? limits.maxPhotoBytes : limits.maxVideoBytes;
  if (sz.size > cap) {
    await purgeStagingDir(staging.stagingDir);
    await markFailed(db, mediaId, 'too_large');
    throw new UploadValidationError(413, 'too_large');
  }

  // Rename raw.bin → original.<ext>
  const originalName = variantFilename('original', sniffed.ext);
  await rename(staging.rawPath, join(staging.stagingDir, originalName));

  // §6.2 step 4 — dedupe by server-authoritative hash against another READY row on the same baby.
  const dupe = db
    .select({ id: media.id })
    .from(media)
    .where(
      and(
        eq(media.babyId, input.babyId),
        eq(media.contentHash, hash),
        eq(media.status, 'ready'),
        ne(media.id, mediaId)
      )
    )
    .get();

  if (dupe) {
    await purgeStagingDir(staging.stagingDir);
    db.transaction((tx) => {
      tx.update(media)
        .set({ status: 'failed', updatedAt: Date.now() })
        .where(eq(media.id, mediaId))
        .run();
      if (input.entryId) {
        tx.insert(entryMedia)
          .values({ entryId: input.entryId, mediaId: dupe.id, attachedBy: input.userId, attachedAt: Date.now() })
          .onConflictDoNothing()
          .run();
      }
    });
    return { kind: 'deduped', mediaId: dupe.id };
  }

  // §6.2 step 5 — process: derive variants in-place under staging.
  db.update(media).set({ status: 'processing', updatedAt: Date.now() }).where(eq(media.id, mediaId)).run();
  try {
    if (sniffed.type === 'photo') {
      await derivePhotoVariants(join(staging.stagingDir, originalName), staging.stagingDir);
    } else {
      await deriveVideoPoster(join(staging.stagingDir, originalName), staging.stagingDir);
    }
  } catch (e) {
    await purgeStagingDir(staging.stagingDir);
    await markFailed(db, mediaId, 'derive_failed');
    throw new UploadValidationError(500, 'derive_failed');
  }

  // §6.2 step 6 — atomic commit + DB row to ready.
  const takenAtMs = sniffed.takenAtMs ?? nowMs;
  let relativePath: string;
  try {
    relativePath = await commitStaging(input.dataDir, {
      mediaId,
      babyId: input.babyId,
      createdAtMs: takenAtMs,
      ext: sniffed.ext,
      kind: sniffed.type
    });
  } catch (e) {
    await purgeStagingDir(staging.stagingDir).catch(() => {});
    await markFailed(db, mediaId, 'commit_failed');
    throw new UploadValidationError(500, 'commit_failed');
  }

  try {
    db.transaction((tx) => {
      tx.update(media)
        .set({
          status: 'ready',
          type: sniffed.type,
          mimeType: sniffed.mimeType,
          sizeBytes: sz.size,
          contentHash: hash,
          width: sniffed.width,
          height: sniffed.height,
          durationSec: sniffed.durationSec,
          relativePath,
          originalExt: sniffed.ext,
          takenAt: takenAtMs,
          updatedAt: Date.now()
        })
        .where(eq(media.id, mediaId))
        .run();
      if (input.entryId) {
        tx.insert(entryMedia)
          .values({ entryId: input.entryId, mediaId, attachedBy: input.userId, attachedAt: Date.now() })
          .onConflictDoNothing()
          .run();
      }
    });
  } catch (e) {
    // §6.2 step 6 last bullet — extreme race: partial-unique conflict on (babyId, contentHash, status=ready).
    // Roll this row to failed; the other request keeps its ready row. Client may retry → next attempt dedupes.
    await markFailed(db, mediaId, 'commit_conflict').catch(() => {});
    throw new UploadValidationError(409, 'commit_conflict');
  }
  return { kind: 'created', mediaId };
}

async function streamHashFile(path: string): Promise<string> {
  return await new Promise<string>((resolve, reject) => {
    const h = createHash('sha256');
    const s = createReadStream(path);
    s.on('data', (c) => h.update(c));
    s.on('end', () => resolve(h.digest('hex')));
    s.on('error', reject);
  });
}

async function markFailed(db: ReturnType<typeof getDb>['db'], mediaId: string, _reason: string) {
  db.update(media).set({ status: 'failed', updatedAt: Date.now() }).where(eq(media.id, mediaId)).run();
}

function sanitizeFilenameDisplay(s: string): string {
  return (s || 'upload')
    .replace(/[/\\ -]/g, '_')
    .slice(0, 255);
}
```

> Note: this orchestrator does not run as a streamed-while-receive hash — it uses formidable's spooled temp file then re-streams from disk. The spec §6.2 admits both shapes; the practical reason for the disk-resident approach is that Next.js App Router request streams interact poorly with re-entrant formidable. The §6.2 invariants are preserved: server computes the hash, server computes the MIME, dedupe only happens after the bytes are on disk.

```ts
// app/api/media/upload/route.ts
import { resolve } from 'node:path';
import { tmpdir } from 'node:os';
import { mkdir } from 'node:fs/promises';
import { join } from 'node:path';
import { IncomingMessage } from 'node:http';
import formidable from 'formidable';
import { Readable } from 'node:stream';
import { withAuthorizedAction } from '@/lib/permissions/action-template';
import { loadAndAssertTarget } from '@/lib/permissions/target-loaders';
import { jsonBadRequest, jsonNotFound, UUID_RE } from '@/lib/permissions/responses';
import { runUploadPipeline, UploadValidationError } from '@/lib/media/upload-pipeline';

export const dynamic = 'force-dynamic';
export const runtime = 'nodejs';

const dataDir = process.env.BABYLOOM_DATA_DIR
  ? resolve(process.env.BABYLOOM_DATA_DIR)
  : resolve(process.cwd(), 'data');

export const POST = withAuthorizedAction({
  action: 'media:write',
  // The full resource isn't known until we parse multipart — defer the assertion to the
  // explicit loadAndAssertTarget calls below. action-template's resolveResource is left
  // out so the wrapper only enforces "logged in"; the targets do the rest.
})(async (req, userId) => {
  let parsed: ParsedMultipart;
  try {
    parsed = await parseMultipart(req);
  } catch (e: any) {
    if (e?.code === 'too_large') return new Response(JSON.stringify({ error: 'too_large' }), { status: 413 });
    return jsonBadRequest('invalid_multipart');
  }

  if (!parsed.babyId || !UUID_RE.test(parsed.babyId)) return jsonBadRequest('babyId_required');
  if (!parsed.clientUploadId || parsed.clientUploadId.length < 8 || parsed.clientUploadId.length > 100) {
    return jsonBadRequest('clientUploadId_required');
  }
  if (parsed.entryId && !UUID_RE.test(parsed.entryId)) return jsonNotFound();
  if (!parsed.file) return jsonBadRequest('file_required');

  // Target #1: babyId. Loader JOINs nothing (babies has no parent); status must be active.
  let baby: any;
  try {
    baby = await loadAndAssertTarget({
      id: parsed.babyId,
      table: 'babies',
      allowedStatuses: ['active'],
      requirePermission: { userId, action: 'media:write' },
      dataDir
    });
  } catch {
    return jsonNotFound();
  }

  // Target #2: entryId (optional). Loader JOINs babies (P2a), enforces baby active + cross-baby check.
  let entryRow: any = null;
  if (parsed.entryId) {
    try {
      entryRow = await loadAndAssertTarget({
        id: parsed.entryId,
        table: 'entries',
        allowedStatuses: ['active'],
        requirePermission: { userId, action: 'entry:write' },
        dataDir
      });
    } catch {
      return jsonNotFound();
    }
    if (entryRow.babyId !== baby.id) return jsonNotFound();
  }

  try {
    const outcome = await runUploadPipeline({
      dataDir,
      userId,
      babyId: baby.id,
      entryId: entryRow ? entryRow.id : null,
      clientUploadId: parsed.clientUploadId,
      filenameRaw: parsed.file.originalFilename || 'upload',
      uploadedFilePath: parsed.file.filepath,
      uploadedSizeBytes: parsed.file.size
    });
    switch (outcome.kind) {
      case 'retry':
        return Response.json({ mediaId: outcome.mediaId, status: outcome.status, pollUrl: outcome.pollUrl }, { status: 202 });
      case 'ready_idempotent':
        return Response.json({ mediaId: outcome.mediaId, deduplicated: false });
      case 'deduped':
        return Response.json({ mediaId: outcome.mediaId, deduplicated: true });
      case 'created':
        return Response.json({ mediaId: outcome.mediaId, deduplicated: false });
    }
  } catch (e) {
    if (e instanceof UploadValidationError) {
      return new Response(JSON.stringify({ error: e.errCode }), { status: e.httpStatus, headers: { 'content-type': 'application/json' } });
    }
    throw e;
  }
});

interface ParsedMultipart {
  babyId?: string;
  entryId?: string;
  clientUploadId?: string;
  file?: formidable.File;
}

async function parseMultipart(req: Request): Promise<ParsedMultipart> {
  const tmp = join(tmpdir(), 'babyloom-upload');
  await mkdir(tmp, { recursive: true });
  const form = formidable({
    uploadDir: tmp,
    keepExtensions: false,
    maxFileSize: 1024 * 1024 * 1024, // 1 GB hard ceiling at the parser; the pipeline re-enforces config caps.
    multiples: false
  });
  const nodeReq = await toNodeIncoming(req);
  return await new Promise((resolve, reject) => {
    form.parse(nodeReq, (err, fields, files) => {
      if (err) return reject(err);
      const get = (k: string): string | undefined => {
        const v = fields[k];
        if (!v) return undefined;
        return Array.isArray(v) ? v[0] : v;
      };
      const f = (files.file as formidable.File[] | formidable.File | undefined);
      const file = Array.isArray(f) ? f[0] : f;
      resolve({
        babyId: get('babyId'),
        entryId: get('entryId'),
        clientUploadId: get('clientUploadId'),
        file
      });
    });
  });
}

// formidable speaks IncomingMessage; bridge from the WHATWG Request.
async function toNodeIncoming(req: Request): Promise<IncomingMessage> {
  const headers = Object.fromEntries(req.headers.entries());
  const readable = Readable.fromWeb(req.body as any);
  const inc = readable as unknown as IncomingMessage;
  (inc as any).headers = headers;
  (inc as any).method = req.method;
  (inc as any).url = '/';
  return inc;
}
```

```ts
// app/api/entries/[entryId]/media/[mediaId]/attach/route.ts
// eslint-route-auth: manual
import { resolve } from 'node:path';
import { and, eq } from 'drizzle-orm';
import { type NextRequest } from 'next/server';
import { getDb } from '@/lib/db/client';
import { entryMedia } from '@/lib/db/schema';
import { getSessionUserId } from '@/lib/permissions/session';
import { loadAndAssertTarget } from '@/lib/permissions/target-loaders';
import { jsonNotFound, jsonUnauthorized, UUID_RE } from '@/lib/permissions/responses';
import { ForbiddenError, NotFoundError, UnauthorizedError } from '@/lib/permissions/errors';

export const runtime = 'nodejs';

const dataDir = process.env.BABYLOOM_DATA_DIR
  ? resolve(process.env.BABYLOOM_DATA_DIR)
  : resolve(process.cwd(), 'data');

interface Ctx {
  params: Promise<{ entryId: string; mediaId: string }>;
}

async function handle(
  req: NextRequest,
  ctx: Ctx,
  op: 'attach' | 'detach'
): Promise<Response> {
  let userId: string;
  try {
    userId = await getSessionUserId(req);
  } catch (e) {
    if (e instanceof UnauthorizedError) return jsonUnauthorized();
    throw e;
  }

  const params = await ctx.params;
  if (!UUID_RE.test(params.entryId) || !UUID_RE.test(params.mediaId)) return jsonNotFound();

  let mediaRow: any, entry: any;
  try {
    mediaRow = await loadAndAssertTarget({
      id: params.mediaId,
      table: 'media',
      allowedStatuses: ['ready'],
      requirePermission: { userId, action: 'media:read' },
      dataDir
    });
    entry = await loadAndAssertTarget({
      id: params.entryId,
      table: 'entries',
      allowedStatuses: ['active'],
      requirePermission: { userId, action: 'entry:write' },
      dataDir
    });
  } catch (e) {
    if (e instanceof ForbiddenError || e instanceof NotFoundError) return jsonNotFound();
    throw e;
  }
  if (entry.babyId !== mediaRow.babyId) return jsonNotFound();

  const { db } = getDb({ dataDir });
  if (op === 'attach') {
    const existing = db
      .select({ entryId: entryMedia.entryId })
      .from(entryMedia)
      .where(and(eq(entryMedia.entryId, entry.id), eq(entryMedia.mediaId, mediaRow.id)))
      .get();
    if (existing) return Response.json({ attached: true, alreadyExisted: true });
    db.insert(entryMedia)
      .values({ entryId: entry.id, mediaId: mediaRow.id, attachedBy: userId, attachedAt: Date.now() })
      .run();
    return Response.json({ attached: true, alreadyExisted: false });
  }
  db.delete(entryMedia)
    .where(and(eq(entryMedia.entryId, entry.id), eq(entryMedia.mediaId, mediaRow.id)))
    .run();
  return Response.json({ detached: true });
}

export const POST = (req: NextRequest, ctx: Ctx) => handle(req, ctx, 'attach');
export const DELETE = (req: NextRequest, ctx: Ctx) => handle(req, ctx, 'detach');
```

```ts
// lib/media/reconcile.ts
import { readdir } from 'node:fs/promises';
import { existsSync } from 'node:fs';
import { join } from 'node:path';
import { and, eq, inArray, lt } from 'drizzle-orm';
import { getDb } from '@/lib/db/client';
import { media } from '@/lib/db/schema';
import { purgeStagingDir } from './storage';
import { stagingDir } from './paths';

const STUCK_PENDING_MS = 60 * 60 * 1000;

export async function runReconcileOnce(opts: { dataDir: string; nowMs: number }): Promise<void> {
  const { db } = getDb({ dataDir: opts.dataDir });

  // 1. Stuck pending/processing → failed.
  db.update(media)
    .set({ status: 'failed', updatedAt: opts.nowMs })
    .where(
      and(
        inArray(media.status, ['pending', 'processing']),
        lt(media.createdAt, opts.nowMs - STUCK_PENDING_MS)
      )
    )
    .run();

  // 2. Orphan staging dirs.
  const stagingRoot = join(opts.dataDir, 'media', '_staging');
  if (!existsSync(stagingRoot)) return;

  let entries: string[];
  try {
    entries = await readdir(stagingRoot);
  } catch {
    return;
  }
  for (const name of entries) {
    const row = db
      .select({ status: media.status })
      .from(media)
      .where(eq(media.id, name))
      .get();
    const keep = row && (row.status === 'pending' || row.status === 'processing');
    if (!keep) {
      await purgeStagingDir(stagingDir(opts.dataDir, name)).catch(() => {});
    }
  }
}

let timer: NodeJS.Timeout | null = null;

export function startReconcileWorker(dataDir: string): void {
  if (timer) return;
  const tick = () => runReconcileOnce({ dataDir, nowMs: Date.now() }).catch(() => {});
  tick();
  timer = setInterval(tick, 24 * 60 * 60 * 1000);
  // Allow process exit even with timer pending (test runners).
  timer.unref?.();
}

export function stopReconcileWorkerForTesting(): void {
  if (timer) clearInterval(timer);
  timer = null;
}
```

```tsx
// components/media/UploadButton.tsx
'use client';
import * as React from 'react';

export interface UploadedMedia {
  mediaId: string;
  filename: string;
  status: 'ready' | 'pending';
}

export interface UploadButtonProps {
  babyId: string;
  onUploaded: (m: UploadedMedia) => void;
  disabled?: boolean;
  multiple?: boolean;
}

const UPLOAD_URL = '/api/media/upload';

export function UploadButton({ babyId, onUploaded, disabled, multiple = true }: UploadButtonProps) {
  const inputRef = React.useRef<HTMLInputElement>(null);
  const [busy, setBusy] = React.useState(false);
  const [error, setError] = React.useState<string | null>(null);

  async function handleFiles(files: FileList) {
    setBusy(true);
    setError(null);
    try {
      for (const file of Array.from(files)) {
        const clientUploadId = crypto.randomUUID();
        const fd = new FormData();
        fd.append('babyId', babyId);
        fd.append('clientUploadId', clientUploadId);
        fd.append('file', file);
        const res = await fetch(UPLOAD_URL, { method: 'POST', body: fd });
        if (!res.ok) {
          const body = await res.json().catch(() => ({}));
          throw new Error(body.error || `http_${res.status}`);
        }
        const json = await res.json();
        if (json.status === 'pending' || json.status === 'processing') {
          onUploaded({ mediaId: json.mediaId, filename: file.name, status: 'pending' });
          await pollUntilReady(json.mediaId);
        }
        onUploaded({ mediaId: json.mediaId, filename: file.name, status: 'ready' });
      }
    } catch (e: any) {
      setError(e.message || 'upload_failed');
    } finally {
      setBusy(false);
      if (inputRef.current) inputRef.current.value = '';
    }
  }

  return (
    <div>
      <input
        ref={inputRef}
        type="file"
        accept="image/jpeg,image/png,image/webp,image/heic,image/heif,video/mp4,video/quicktime"
        multiple={multiple}
        hidden
        onChange={(e) => e.target.files && handleFiles(e.target.files)}
      />
      <button
        type="button"
        disabled={disabled || busy}
        onClick={() => inputRef.current?.click()}
      >
        {busy ? 'Uploading…' : 'Add photo / video'}
      </button>
      {error && <p role="alert">{error}</p>}
    </div>
  );
}

async function pollUntilReady(mediaId: string): Promise<void> {
  const deadline = Date.now() + 5 * 60 * 1000;
  while (Date.now() < deadline) {
    await new Promise((r) => setTimeout(r, 1500));
    const r = await fetch(`/api/media/${mediaId}/status`);
    if (!r.ok) throw new Error('status_unavailable');
    const j = await r.json();
    if (j.status === 'ready') return;
    if (j.status === 'failed') throw new Error('processing_failed');
  }
  throw new Error('upload_timeout');
}
```
